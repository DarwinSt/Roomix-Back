package com.example.roomix.common.exception;

import org.springframework.http.HttpStatus;

/**
 * Excepción de negocio cuyo mensaje y código HTTP provienen de un enum.
 */
public abstract class BusinessException extends RuntimeException {

    private final String errorCode;
    private final HttpStatus httpStatus;

    protected BusinessException(String errorCode, HttpStatus httpStatus, String message) {
        super(message);
        this.errorCode = errorCode;
        this.httpStatus = httpStatus;
    }

    public String getErrorCode() {
        return errorCode;
    }

    public HttpStatus getHttpStatus() {
        return httpStatus;
    }

    /** Título del ProblemDetail (nombre del código de error). Los módulos nuevos lo sobrescriben. */
    public String getTitulo() {
        return errorCode;
    }

    /** Módulo funcional que originó el error (ej. FINANZAS). */
    public String getModulo() {
        return null;
    }
}
